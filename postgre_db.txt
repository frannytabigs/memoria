-- ============================================================================
-- memoria_db — PostgreSQL 18 conversion of the MySQL schema
-- ============================================================================
--
-- HOW TO RUN:
--   1. Connect to a maintenance DB (postgres / template1):
--        DROP DATABASE IF EXISTS memoria_db;
--        CREATE DATABASE memoria_db;
--   2. Reconnect to memoria_db (e.g. \c memoria_db in psql).
--   3. Paste this entire file.
--
-- Everything below assumes you're connected to memoria_db.
--
-- Preserved from the MySQL version:
--   * all constraint names the PHP code relies on (uk_active_* / chk_*)
--   * all shadow columns and their UNIQUE indexes
--   * all foreign-key ON DELETE behavior
--   * all indexes, functionally equivalent
--   * the two seed users / settings / payments rows
--
-- Rewritten for PostgreSQL:
--   * AUTO_INCREMENT → GENERATED BY DEFAULT AS IDENTITY
--   * ENUM(...) → VARCHAR + CHECK (same accepted values, same case-sensitivity)
--   * ON UPDATE CURRENT_TIMESTAMP → trigger_set_updated_at() trigger
--   * DATETIME → TIMESTAMP
--   * backticks → double quotes (only needed for reserved words) or bare names
--   * prefix indexes → left(col, N) functional indexes
--   * ALTER TABLE … COMMENT → COMMENT ON TABLE
--   * NULL in identity column → omit the column
-- ============================================================================

-- ----------------------------------------------------------------------------
-- Idempotent clean slate: drop tables in reverse dependency order
-- ----------------------------------------------------------------------------
DROP TABLE IF EXISTS placement_history CASCADE;
DROP TABLE IF EXISTS reservations      CASCADE;
DROP TABLE IF EXISTS interments        CASCADE;
DROP TABLE IF EXISTS graves            CASCADE;
DROP TABLE IF EXISTS blocks            CASCADE;
DROP TABLE IF EXISTS payments          CASCADE;
DROP TABLE IF EXISTS settings          CASCADE;
DROP TABLE IF EXISTS users             CASCADE;

DROP FUNCTION IF EXISTS trigger_set_updated_at() CASCADE;

-- ----------------------------------------------------------------------------
-- Shared trigger function: replaces MySQL's ON UPDATE CURRENT_TIMESTAMP
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION trigger_set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.updated_at IS NOT DISTINCT FROM OLD.updated_at THEN
        NEW.updated_at = CURRENT_TIMESTAMP;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================================================================
-- USERS (must be first — every other table references it)
-- ============================================================================
CREATE TABLE users (
    user_id       INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    -- PostgreSQL is case-sensitive by default for `=` on VARCHAR/text,
    -- so this replaces utf8mb4_bin without any collation clause.
    username      VARCHAR(50)  NOT NULL,
    role          VARCHAR(20)  NOT NULL DEFAULT 'Grounds Staff'
        CHECK (role IN ('Administrator', 'Office Staff', 'Grounds Staff')),
    status        VARCHAR(20)  NOT NULL DEFAULT 'Unverified'
        CHECK (status IN ('Verified', 'Unverified')),
    email         VARCHAR(255) NOT NULL,
    phone_number  VARCHAR(20)  NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name          VARCHAR(255) NOT NULL,
    session_token TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    created_by INT,
    updated_by INT,

    -- Shadow columns: unique only among ACTIVE rows (deleted_at IS NULL).
    -- NULL values don't conflict in a UNIQUE index, so soft-deleted rows
    -- become free to re-use the same username/email/phone.
    active_username VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN username ELSE NULL END) STORED,
    active_email    VARCHAR(255)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN email ELSE NULL END) STORED,
    active_phone    VARCHAR(20)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN phone_number ELSE NULL END) STORED,

    CONSTRAINT uk_active_username UNIQUE (active_username),
    CONSTRAINT uk_active_email    UNIQUE (active_email),
    CONSTRAINT uk_active_phone    UNIQUE (active_phone),

    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER set_updated_at BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();

-- ============================================================================
-- BLOCKS
-- ============================================================================
CREATE TABLE blocks (
    block_id    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    block_name  VARCHAR(100) NOT NULL,
    block_type  VARCHAR(30)  NOT NULL
        CHECK (block_type IN (
            'Niche', 'Bone Chamber', 'Lawn/Grounds',
            'Unmapped Area', 'Private', 'Mausoleum',
            'Mass Grave', 'Cluster', 'Block'
        )),
    image_link  TEXT,
    coordinates JSON,
    rows        INT NOT NULL DEFAULT 0,
    cols        INT NOT NULL DEFAULT 0,
    rotation    DECIMAL(10,6) DEFAULT 0.000000,
    remarks     TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    created_by INT,
    updated_by INT,

    active_block_name VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN block_name ELSE NULL END) STORED,
    CONSTRAINT uk_active_block_name UNIQUE (active_block_name),

    FOREIGN KEY (created_by) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER set_updated_at BEFORE UPDATE ON blocks
    FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();

-- ============================================================================
-- GRAVES

CREATE TABLE graves (
    grave_id   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    block_id   INT NOT NULL,
    grave_code VARCHAR(50) NOT NULL,
    row_num    INT,
    col_num    INT,
    status     VARCHAR(20) DEFAULT 'Vacant'
        CHECK (status IN ('Vacant', 'Occupied', 'Other')),
    remarks    TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    created_by INT,
    updated_by INT,

    active_grave_code VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN grave_code ELSE NULL END) STORED,
    CONSTRAINT uk_active_grave_code UNIQUE (active_grave_code),

    FOREIGN KEY (block_id)   REFERENCES blocks(block_id) ON DELETE RESTRICT,
    FOREIGN KEY (created_by) REFERENCES users(user_id)   ON DELETE SET NULL,
    FOREIGN KEY (updated_by) REFERENCES users(user_id)   ON DELETE SET NULL
);

CREATE TRIGGER set_updated_at BEFORE UPDATE ON graves
    FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();

-- ============================================================================
-- INTERMENTS (PRESENT — current state)
-- ============================================================================
CREATE TABLE interments (
    interment_id  INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    control_number VARCHAR(100) NOT NULL,

    deceased_name         VARCHAR(255) NOT NULL,
    last_known_address    TEXT,
    death_certificate     VARCHAR(100),
    deceased_date_of_birth DATE,
    deceased_date_of_death DATE,
    deceased_sex          VARCHAR(20) DEFAULT 'Unknown'
        CHECK (deceased_sex IN ('Male', 'Female', 'Unknown')),

    current_grave_id INT,

    contact_person_name             VARCHAR(255),
    contact_person_phone_number     VARCHAR(50),
    contact_person_email            VARCHAR(150),
    contact_person_address_barangay TEXT,
    contact_person_address          TEXT,

    assistance_type VARCHAR(50) NOT NULL DEFAULT 'Burial'
        CHECK (assistance_type IN (
            'Burial', 'Transfer the remains of the late', 'Other'
        )),

    burial_permit_number     VARCHAR(100),
    burial_permit_date       DATE,
    transfer_permit_number   VARCHAR(100),
    transfer_permit_issued_by VARCHAR(150),
    transfer_permit_date     DATE,
    exhumation_permit_number VARCHAR(100),
    exhumation_permit_date   DATE,

    date_buried           DATE,
    date_exhumed          DATE,
    burial_clearance_date DATE,
    lease_expiration_date DATE,

    status VARCHAR(20) NOT NULL DEFAULT 'Pending'
        CHECK (status IN ('Pending', 'Active', 'Inactive')),

    remarks TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    created_by INT,
    updated_by INT,

    active_control_number VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN control_number ELSE NULL END) STORED,
    CONSTRAINT uk_active_control_number UNIQUE (active_control_number),

    FOREIGN KEY (current_grave_id) REFERENCES graves(grave_id) ON DELETE RESTRICT,
    FOREIGN KEY (created_by)       REFERENCES users(user_id)   ON DELETE SET NULL,
    FOREIGN KEY (updated_by)       REFERENCES users(user_id)   ON DELETE SET NULL,

    -- Named exactly as the PHP code expects (reserve.php / records.php grep for this)
    CONSTRAINT chk_status_grave CHECK (
        status = 'Active' OR current_grave_id IS NULL
    )
);

CREATE TRIGGER set_updated_at BEFORE UPDATE ON interments
    FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();

-- ============================================================================
-- RESERVATIONS (FUTURE — planned state, frozen after creation)
-- ============================================================================
CREATE TABLE reservations (
    reservation_id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    pending_interment_id INT NOT NULL,
    target_grave_id      INT NOT NULL,
    old_interment_id     INT,

    old_new_grave_id          INT,
    old_new_status            VARCHAR(20)
        CHECK (old_new_status IN ('Active', 'Inactive')),
    old_new_assistance_type   VARCHAR(50)
        CHECK (old_new_assistance_type IN (
            'Burial', 'Transfer the remains of the late', 'Other'
        )),
    old_new_remarks           TEXT,

    exhumation_permit_number VARCHAR(100),
    transfer_permit_number   VARCHAR(100),

    reservation_notes TEXT,

    reservation_state VARCHAR(20) NOT NULL DEFAULT 'Active'
        CHECK (reservation_state IN ('Active', 'Executed', 'Cancelled')),
    actioned_at TIMESTAMP,
    actioned_by INT,

    old_edited_fields JSON,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INT,
    updated_by INT,

    -- Shadow columns: uniqueness only among Active reservations
    active_pending_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN pending_interment_id ELSE NULL END) STORED,
    active_target_grave_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN target_grave_id ELSE NULL END) STORED,
    active_old_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN old_interment_id ELSE NULL END) STORED,

    CONSTRAINT uk_active_pending_interment UNIQUE (active_pending_interment_id),
    CONSTRAINT uk_active_target_grave      UNIQUE (active_target_grave_id),
    CONSTRAINT uk_active_old_interment     UNIQUE (active_old_interment_id),

    FOREIGN KEY (pending_interment_id) REFERENCES interments(interment_id) ON DELETE RESTRICT,
    FOREIGN KEY (target_grave_id)      REFERENCES graves(grave_id)          ON DELETE RESTRICT,
    FOREIGN KEY (old_interment_id)     REFERENCES interments(interment_id)  ON DELETE RESTRICT,
    FOREIGN KEY (old_new_grave_id)     REFERENCES graves(grave_id)          ON DELETE RESTRICT,
    FOREIGN KEY (actioned_by)          REFERENCES users(user_id)            ON DELETE SET NULL,
    FOREIGN KEY (created_by)           REFERENCES users(user_id)            ON DELETE SET NULL,
    FOREIGN KEY (updated_by)           REFERENCES users(user_id)            ON DELETE SET NULL,

    CONSTRAINT chk_old_pairing CHECK (
        (old_interment_id IS NULL
            AND old_new_grave_id IS NULL
            AND old_new_status IS NULL
            AND old_new_assistance_type IS NULL)
        OR
        (old_interment_id IS NOT NULL AND old_new_status IS NOT NULL)
    ),

    CONSTRAINT chk_inactive_has_no_grave CHECK (
        NOT (old_new_status = 'Inactive' AND old_new_grave_id IS NOT NULL)
    ),

    CONSTRAINT chk_old_new_grave_not_target CHECK (
        old_new_grave_id IS NULL OR old_new_grave_id <> target_grave_id
    )
);

CREATE TRIGGER set_updated_at BEFORE UPDATE ON reservations
    FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();

-- ============================================================================
-- PLACEMENT_HISTORY (PAST — append-only)
-- ============================================================================
CREATE TABLE placement_history (
    history_id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    interment_id INT NOT NULL,
    event_type   VARCHAR(20) NOT NULL
        CHECK (event_type IN ('Initial', 'Move', 'Removal')),

    from_grave_id INT,
    to_grave_id   INT,

    event_date TIMESTAMP NOT NULL,
    reason     TEXT,

    related_interment_id INT,
    reservation_id       INT,

    snapshot_before JSON,
    snapshot_after  JSON,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INT,

    FOREIGN KEY (interment_id)         REFERENCES interments(interment_id)      ON DELETE RESTRICT,
    FOREIGN KEY (from_grave_id)        REFERENCES graves(grave_id)              ON DELETE RESTRICT,
    FOREIGN KEY (to_grave_id)          REFERENCES graves(grave_id)              ON DELETE RESTRICT,
    FOREIGN KEY (related_interment_id) REFERENCES interments(interment_id)      ON DELETE SET NULL,
    FOREIGN KEY (reservation_id)       REFERENCES reservations(reservation_id)  ON DELETE SET NULL,
    FOREIGN KEY (created_by)           REFERENCES users(user_id)                ON DELETE SET NULL
);

CREATE TABLE payments (
    payment_id       INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    reference_number VARCHAR(100) NOT NULL,
    payment_channel  VARCHAR(100) NOT NULL,
    amount           DECIMAL(10,2) NOT NULL,
    deceased_name    VARCHAR(255),
    payers_phone_number VARCHAR(20),
    payers_email     VARCHAR(255),
    payers_name      VARCHAR(255),

    purpose    VARCHAR(100) NOT NULL,
    image_link TEXT,

    confirmed_office_staff INT,
    confirmed_ground_staff INT,
    remarks_payer   TEXT,
    remarks_office  TEXT,
    remarks_grounds TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    created_by INT,
    updated_by INT,

    active_reference_number VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN reference_number ELSE NULL END) STORED,
    CONSTRAINT uk_active_reference_number UNIQUE (active_reference_number),

    FOREIGN KEY (confirmed_office_staff) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (confirmed_ground_staff) REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (created_by)             REFERENCES users(user_id) ON DELETE SET NULL,
    FOREIGN KEY (updated_by)             REFERENCES users(user_id) ON DELETE SET NULL
);
