CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT, 
    username VARCHAR(50) UNIQUE NOT NULL,
    role ENUM('Administrator', 'Staff') NOT NULL DEFAULT 'Staff',
    status ENUM('Verified', 'Unverified') NOT NULL DEFAULT 'Unverified',
    email VARCHAR(255) UNIQUE NOT NULL,
    phone_number VARCHAR(20) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(50) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE settings (
    setting_id INT AUTO_INCREMENT PRIMARY KEY,
    setting_key VARCHAR(50) NOT NULL UNIQUE,
    setting_value TEXT,
    description VARCHAR(255) NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- 1. CONTACTS TABLE (The living family members, guardians, or CSWS officers)
CREATE TABLE contacts (
    contact_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255), -- Nullable in case the agency doesn't provide a specific person's name
    address TEXT,
    phone_number VARCHAR(20),
    email_address VARCHAR(100), 
    remarks TEXT -- Good place to put "City Social Welfare Services"
);

-- 2A. BLOCKS TABLE (The big areas/buildings)
CREATE TABLE blocks (
    block_id INT AUTO_INCREMENT PRIMARY KEY,
    block_name VARCHAR(100) UNIQUE NOT NULL, 
    block_type ENUM('Niche Wall', 'Bone Chamber Building', 'Lawn/Grounds', 'Unmapped Area'), -- Added 'Unmapped Area' just in case!
    total_rows INT,    
    total_columns INT, 
    remarks TEXT
);

-- 2B. GRAVES TABLE (The individual holes/slots inside a block)
CREATE TABLE graves (
    grave_id INT AUTO_INCREMENT PRIMARY KEY,
    block_id INT, 
    grave_code VARCHAR(50) UNIQUE, 
    row_num INT,         
    col_num INT,         
    floor_level INT,     
    status ENUM('Vacant', 'Occupied', 'Reserved', 'Pending Exhumation', 'Under Maintenance', 'Merged') DEFAULT 'Vacant', -- ADDED 'Merged'
    merged_with_id INT,  -- NEW: Tells us which grave swallowed this one up    
    remarks TEXT,
    FOREIGN KEY (block_id) REFERENCES blocks(block_id) ON DELETE RESTRICT,
    FOREIGN KEY (merged_with_id) REFERENCES graves(grave_id) ON DELETE SET NULL -- This links back to its own table!
);

-- 3. DECEASED TABLE (Biological/identifying info)
CREATE TABLE deceased (
    deceased_id INT AUTO_INCREMENT PRIMARY KEY,
    control_number VARCHAR(100) UNIQUE NOT NULL,
    name VARCHAR(100), -- Nullable for unidentified individuals
    sex ENUM('Male', 'Female', 'Unknown') DEFAULT 'Unknown',
    date_of_birth DATE, -- Nullable if unknown
    date_of_death DATE, -- Nullable if unknown
    remarks TEXT -- e.g., "Unidentified male, handled by CSWS"
);

-- 4. INTERMENTS TABLE (The Burial Record)
CREATE TABLE interments (
    interment_id INT AUTO_INCREMENT PRIMARY KEY,
    deceased_id INT NOT NULL, -- There is always a physical body
    grave_id INT,             -- FIXED: Points to graves, NOT blocks. Nullable for corner burials!
    contact_id INT,           -- FIXED: Nullable for homeless/no family cases.
    date_buried DATE,         -- Nullable just in case the exact historical date is lost
    date_exhumed DATE, 
    lease_expiration_date DATE, 
    status ENUM('Active', 'Expired', 'Exhumed', 'Moved to Family') DEFAULT 'Active',
    remarks TEXT,
    FOREIGN KEY (deceased_id) REFERENCES deceased(deceased_id),
    FOREIGN KEY (grave_id) REFERENCES graves(grave_id), 
    FOREIGN KEY (contact_id) REFERENCES contacts(contact_id) ON DELETE SET NULL
);

-- 5. RESERVATIONS TABLE (Reserving vacant spots before death)
CREATE TABLE reservations (
    reservation_id INT AUTO_INCREMENT PRIMARY KEY,
    grave_id INT NOT NULL,    -- FIXED: You must reserve a specific grave, not a whole block.
    contact_id INT, -- could be from csws or no contact and all...
    reservation_date DATE,
    expiration_date DATE, 
    status ENUM('Active', 'Fulfilled', 'Cancelled') DEFAULT 'Active',
    remarks TEXT,
    FOREIGN KEY (grave_id) REFERENCES graves(grave_id), 
    FOREIGN KEY (contact_id) REFERENCES contacts(contact_id)
);

-- 6. SMS NOTIFICATIONS TABLE (Audit trail for the TextBee API)
CREATE TABLE sms_notifications (
    notification_id INT AUTO_INCREMENT PRIMARY KEY,
    interment_id INT,           -- Which burial this relates to
    contact_id INT,             -- The family member receiving the text
    user_id INT,                -- The Staff/Admin who clicked the "Send" button
    phone_number VARCHAR(20),   -- Stored explicitly in case the contact changes their number later
    message_type ENUM('1st Warning', 'Final Warning', 'Expired', 'Custom') DEFAULT '1st Warning',
    message_content TEXT,       -- The exact text that was generated and sent
    status ENUM('Pending', 'Sent', 'Failed') DEFAULT 'Pending', -- Crucial for TextBee tracking
    sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    remarks TEXT,
    FOREIGN KEY (interment_id) REFERENCES interments(interment_id) ON DELETE SET NULL,
    FOREIGN KEY (contact_id) REFERENCES contacts(contact_id) ON DELETE SET NULL,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE RESTRICT
);
