-- ==========================================
-- 2. ENTITY TABLES (People & Places)
-- ==========================================

-- CONTACTS: The living individuals interacting with the cemetery (Requesting Party).
CREATE TABLE contacts (
    contact_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255),
    address TEXT,                -- Specifics like Zone Ahus or Lub-ang ....
    barangay VARCHAR(100),       -- Specifically for the "resident of [ ] Barangay" form requirement
    phone_number VARCHAR(20),
    email_address VARCHAR(100), 
    remarks TEXT, 
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- DECEASED: The biological record of the buried individual. 
-- This is permanent and separate from the burial event itself.
CREATE TABLE deceased (
    deceased_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255), 
    sex ENUM('Male', 'Female', 'Unknown') DEFAULT 'Unknown',
    date_of_birth DATE, 
    date_of_death DATE, 
    death_certificate VARCHAR(100), -- The death certificate attached from the form for identification when needed
    last_known_address TEXT,
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- BLOCKS: The macro-locations (e.g., Niche Wall A, Bone Chamber B).
CREATE TABLE blocks (
    block_id INT AUTO_INCREMENT PRIMARY KEY,
    block_name VARCHAR(100) UNIQUE NOT NULL, 
    block_type ENUM('Niche Wall', 'Bone Chamber', 'Lawn/Grounds', 'Unmapped Area', 'Private/Owned'),
    total_rows INT,    
    total_columns INT, 
    max_capacity INT,
    area_sqm DECIMAL(10,2),
    owner_contact_id INT,        -- Links to contacts if a family owns a private mausoleum
    coordinates JSON,            -- For mapping visuals
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (owner_contact_id) REFERENCES contacts(contact_id) ON DELETE SET NULL
);

-- GRAVES: The micro-locations (individual slots) inside a Block.
CREATE TABLE graves (
    grave_id INT AUTO_INCREMENT PRIMARY KEY,
    block_id INT, 
    grave_code VARCHAR(50) UNIQUE, 
    row_num INT,         
    col_num INT,         
    floor_level INT,     
    status ENUM('Vacant', 'Occupied', 'Reserved', 'Pending Exhumation', 'Under Maintenance', 'Merged') DEFAULT 'Vacant',
    merged_with_id INT,          -- Self-referencing ID if walls are knocked down to combine graves
    coordinates JSON,            -- If stand alone or unmapped or something
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (block_id) REFERENCES blocks(block_id) ON DELETE RESTRICT,
    FOREIGN KEY (merged_with_id) REFERENCES graves(grave_id) ON DELETE SET NULL 
);


-- ==========================================
-- 3. TRANSACTION TABLES (The Core Logic)
-- ==========================================

-- INTERMENTS: Acts as the digital version of the "Burial Clearance" form.
-- Connects a Deceased, a Contact, and a Grave together for a specific time period.
CREATE TABLE interments (
    interment_id INT AUTO_INCREMENT PRIMARY KEY,
    control_number VARCHAR(100) UNIQUE NOT NULL, 
    deceased_id INT NOT NULL, 
    grave_id INT,             
    contact_id INT,           
    
    -- Document Checkboxes
    assistance_type ENUM('Burial', 'Transfer', 'Other') NOT NULL DEFAULT 'Burial',
    assistance_other_remarks VARCHAR(255), 
    
    -- Required Document Tracking
    burial_permit_no VARCHAR(100),
    burial_permit_date DATE,
    transfer_permit_no VARCHAR(100),
    transfer_permit_issued_by VARCHAR(150),
    transfer_permit_date DATE,
    exhumation_permit_no VARCHAR(100),
    exhumation_permit_date DATE,
    
    -- Important Dates
    date_buried DATE,            
    date_exhumed DATE, 
    clearance_date DATE,         
    lease_expiration_date DATE,  
    
    status ENUM('Active', 'Expired', 'Exhumed', 'Moved to Family') DEFAULT 'Active',
    
    -- SIMPLIFIED SMS TRACKING
    sms_status ENUM('Not Sent', 'Warning Sent', 'Expired Notice Sent') DEFAULT 'Not Sent',
    last_sms_sent_at TIMESTAMP NULL,
    
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (deceased_id) REFERENCES deceased(deceased_id) ON DELETE RESTRICT,
    FOREIGN KEY (grave_id) REFERENCES graves(grave_id) ON DELETE RESTRICT, 
    FOREIGN KEY (contact_id) REFERENCES contacts(contact_id) ON DELETE RESTRICT
);

-- RESERVATIONS: Holds a grave slot before a burial occurs.
CREATE TABLE reservations (
    reservation_id INT AUTO_INCREMENT PRIMARY KEY,
    grave_id INT NOT NULL,    
    contact_id INT, 
    reservation_date DATE,
    expiration_date DATE, 
    status ENUM('Active', 'Fulfilled', 'Cancelled') DEFAULT 'Active',
    remarks TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (grave_id) REFERENCES graves(grave_id) ON DELETE RESTRICT, 
    FOREIGN KEY (contact_id) REFERENCES contacts(contact_id) ON DELETE RESTRICT
);


-- ==========================================
-- 4. PERFORMANCE INDEXES
-- ==========================================
-- Indexes optimize database read times. They act like a table of contents in a book.

-- Interments Table Indexes
CREATE INDEX idx_lease_expiration ON interments(lease_expiration_date); -- Speeds up your daily expiring lease checks
CREATE INDEX idx_interment_status ON interments(status);                -- Speeds up queries filtering by 'Active' vs 'Exhumed'
CREATE INDEX idx_control_number ON interments(control_number);          -- Speeds up specific form lookups

-- Graves & Blocks Indexes
CREATE INDEX idx_grave_status ON graves(status);                        -- Speeds up finding 'Vacant' slots for the map
CREATE INDEX idx_block_type ON blocks(block_type);                      -- Speeds up filtering between Niche Walls and Bone Chambers

-- People Indexes
CREATE INDEX idx_deceased_name ON deceased(name);                       -- Crucial for the public/staff search function
CREATE INDEX idx_contact_name ON contacts(name);                        -- Speeds up finding family records

-- ==============================================================================
-- QUERY for: Vacant, Expired, and Expiring (30 Days)
-- ==============================================================================

SELECT 
    -- 1. Get the Vacant Graves
    g.grave_code,
    b.block_name,
    'Vacant' AS urgency_level,           -- Hardcoded tag for the frontend
    NULL AS deceased_name,               -- No body yet
    NULL AS family_contact,              -- No family to contact
    NULL AS lease_expiration_date,       -- No lease
    NULL AS sms_status                   -- No SMS needed
FROM graves g
JOIN blocks b ON g.block_id = b.block_id
WHERE g.status = 'Vacant'

UNION ALL

SELECT 
    -- 2. Get the Expired & Expiring Interments
    g.grave_code,
    b.block_name,
    CASE 
        WHEN i.lease_expiration_date < CURDATE() THEN 'Expired'
        ELSE 'Expiring Soon'
    END AS urgency_level,                -- Dynamically tags based on today's date
    d.name AS deceased_name,
    c.phone_number AS family_contact,
    i.lease_expiration_date,
    i.sms_status                         -- Pulls the simplified SMS tracking we just made!
FROM interments i
JOIN graves g ON i.grave_id = g.grave_id
JOIN blocks b ON g.block_id = b.block_id
JOIN deceased d ON i.deceased_id = d.deceased_id
LEFT JOIN contacts c ON i.contact_id = c.contact_id 
WHERE i.status = 'Active' 
  -- Grabs anything that has already expired OR will expire within the next 30 days
  AND i.lease_expiration_date <= DATE_ADD(CURDATE(), INTERVAL 30 DAY)

-- 3. The Sorting Magic (Most urgent at the top)
ORDER BY 
    CASE urgency_level
        WHEN 'Expired' THEN 1            -- Action required immediately
        WHEN 'Expiring Soon' THEN 2      -- Needs a warning text
        WHEN 'Vacant' THEN 3             -- Just informative, no action needed
    END ASC,
    -- Within the Expired/Expiring groups, this sorts the oldest dates to the very top
    lease_expiration_date ASC;
